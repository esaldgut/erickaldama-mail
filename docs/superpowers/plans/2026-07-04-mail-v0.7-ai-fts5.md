# v0.7 — FTS5 en los AI tools Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** que el tool de búsqueda del agente AI (`internal/aiassist`) use FTS5 (el cache de v0.5) por sender+subject con ranking, cayendo a substring sobre DynamoDB solo si el cache falla — el agente nunca queda ciego.

**Architecture:** una interfaz `Searcher` con un compuesto `fallbackSearcher` (campo `ftsSearch func` inyectable → FTS5; si es nil o devuelve error de infra → un `substringSearcher` sobre `*mailbox.Reader`). `execTool` recibe un `toolDeps{reader, searcher}` y el tool `search_messages` (renombrado de `search_subject`) delega en `deps.searcher`. El wire de `mail ai agent` abre el cache best-effort y construye el compuesto.

**Tech Stack:** Go 1.26.4 · `internal/cache` (FTS5, v0.5) · `internal/mailbox` (Reader DynamoDB/S3) · `internal/redact`. Sin deps nuevas.

## Global Constraints

Copiados verbatim del spec `2026-07-04-mail-v0.7-ai-fts5-design.md`. Cada tarea los hereda:

- **AR-1 (crítico):** `fallbackSearcher.Search` cae a substring **solo si el fts backend es nil o devuelve `error != nil`**. Un `([]Header{}, nil)` (0 resultados) se devuelve tal cual, NUNCA cae a substring.
- **AR-2:** `execTool` recibe un `Searcher` opaco vía `toolDeps`; no conoce `ftsSearcher`/`substringSearcher`/el cache. La construcción vive en el wire.
- **AR-3:** los warnings del fallback y del wire van a **stderr** (`os.Stderr`), nunca al contexto del LLM.
- **AR-4:** `Searcher.Search` lleva `ctx context.Context` uniforme (el substring lo usa para `reader.List`; el FTS lo ignora).
- **CT-1:** el tool se llama `search_messages` (ya no `search_subject`); busca sender+subject; description: "Search messages by sender or subject (full-text, ranked best-first, prefix match). Returns matching headers as JSON."

- **CT-4:** query cruda, sin escapar por el tool (`cache.Search` aplica `EscapeFTS` internamente; el substring hace `ToLower`+`Contains`).

package aiassist

import (
	"context"
	"strings"

	"erickaldama-mail/internal/mailbox"
)

// Searcher busca correos por sender/subject y devuelve headers. La firma lleva ctx porque el substring
// backend lo necesita para reader.List; el FTS lo ignora, pero la firma se mantiene uniforme (AR-4).
type Searcher interface {
	Search(ctx context.Context, mailbox, query string, limit int) ([]mailbox.Header, error)
}

// substringSearcher — el comportamiento histórico (fallback): reader.List(limit) + Contains sobre Subject.
type substringSearcher struct{ r *mailbox.Reader }

func (s *substringSearcher) Search(ctx context.Context, mailbox, query string, limit int) ([]mailbox.Header, error) {
	if s.r == nil {
		return nil, nil
	}
	hs, _, err := s.r.List(ctx, mailbox, int32(limit), nil)
	if err != nil {
		return nil, err
	}
	var hits []mailbox.Header
	lq := strings.ToLower(query)
	for _, h := range hs {
		// stdlib case-insensitive contains — handles Unicode (e.g. "café"); NO hand-rolled toLower (audit).
		if strings.Contains(strings.ToLower(h.Subject), lq) {
			hits = append(hits, h)
		}
	}
	return hits, nil
}
```

import (
	"context"
	"fmt"
	"os"
	"strings"

	"erickaldama-mail/internal/mailbox"
)

// fallbackSearcher — FTS5 si el backend funciona, substring SOLO ante error de infra (AR-1).
// Un resultado de 0 filas sin error NO dispara fallback (es una respuesta válida "sin coincidencias").
// ftsSearch es inyectable: en producción envuelve cache.Search; nil → siempre substring; en tests, un fake.
type fallbackSearcher struct {
	ftsSearch func(mb, q string, n int) ([]mailbox.Header, error)
	sub       Searcher // substringSearcher en producción; recordingSub en tests
}

func (f *fallbackSearcher) Search(ctx context.Context, mailbox, query string, limit int) ([]mailbox.Header, error) {
	if f.ftsSearch != nil {
		hs, err := f.ftsSearch(mailbox, query, limit)
		if err == nil {
			return hs, nil // incluye 0 resultados = respuesta válida (AR-1)
		}
		// err REAL de infra (DB corrupta/bloqueada) → log a stderr (AR-3, nunca al LLM) + cae a substring.
		fmt.Fprintf(os.Stderr, "warning: FTS5 search failed, falling back to substring: %v\n", err)
	}
	return f.sub.Search(ctx, mailbox, query, limit)
}
```

// ftsSearchFromCache adapts (*cache.Cache).Search to the fallbackSearcher.ftsSearch closure shape.
// cache.Search already applies EscapeFTS internally (CT-4), so the raw query is passed through.
func ftsSearchFromCache(c *cache.Cache) func(mb, q string, n int) ([]mailbox.Header, error) {
	return func(mb, q string, n int) ([]mailbox.Header, error) {
		return c.Search(mb, q, n)
	}
}

// NewFallbackSearcher builds the composite Searcher. nil-safe: c==nil → the fts backend is nil → the
// composite always uses substring (AR-1, ER-1). The substring backend always wraps the live Reader.
func NewFallbackSearcher(c *cache.Cache, r *mailbox.Reader) Searcher {
	var fts func(mb, q string, n int) ([]mailbox.Header, error)
	if c != nil {
		fts = ftsSearchFromCache(c)
	}
	return &fallbackSearcher{ftsSearch: fts, sub: &substringSearcher{r: r}}
}

// toolDeps carries what execTool needs: the live Reader (list/read tools) and the Searcher (search tool).
type toolDeps struct {
	reader   *mailbox.Reader
	searcher Searcher
}

// execTool runs one read-only tool.
func execTool(ctx context.Context, deps toolDeps, mb string, call ToolCall) string {
	switch call.Name {
	case "list_messages":
		if deps.reader == nil {
			return "[]"
		}
		limit := int32(20)
		if v, ok := call.Args["limit"].(float64); ok && v > 0 {
			limit = int32(v)
		}
		hs, _, err := deps.reader.List(ctx, mb, limit, nil)
		if err != nil {
			return fmt.Sprintf("error: %v", err)
		}
		b, _ := json.Marshal(hs)
		return string(b)
	case "read_message":
		if deps.reader == nil {
			return "[]"
		}
		key, _ := call.Args["s3Key"].(string)
		rc, err := deps.reader.Open(ctx, key)
		if err != nil {
			return fmt.Sprintf("error: %v", err)
		}
		defer rc.Close()
		body, err := io.ReadAll(io.LimitReader(rc, 256*1024))
		if err != nil {
			return fmt.Sprintf("error: %v", err)
		}
		// Redact at the point the raw mail body leaves the mailbox, BEFORE it re-enters the agent loop.
		return redact.Redact(string(body))
	case "search_messages":
		if deps.searcher == nil {
			return "[]"
		}
		q, _ := call.Args["query"].(string)
		limit := 50
		if l, ok := call.Args["limit"].(float64); ok && l > 0 {
			limit = int(l)
		}
		hs, err := deps.searcher.Search(ctx, mb, q, limit)
		if err != nil {
			return fmt.Sprintf("error: %v", err)
		}
		b, _ := json.Marshal(hs)
		return string(b)
	}
	return fmt.Sprintf("unknown tool %q", call.Name)
}

func RunAgent(ctx context.Context, p LLMProvider, reader *mailbox.Reader, searcher Searcher, mb, goal string, maxIters int) (string, error) {
	if maxIters <= 0 {
		return "", fmt.Errorf("RunAgent: maxIters must be > 0, got %d", maxIters)
	}
	deps := toolDeps{reader: reader, searcher: searcher}
	msgs := []Message{{Role: "user", Content: goal}}
	tools := ReadOnlyTools()
	for i := 0; i < maxIters; i++ {
		resp, err := p.Chat(ctx, msgs, tools)
		if err != nil {
			return "", err
		}
		if resp.Stop != "tool_use" || len(resp.ToolCalls) == 0 {
			return resp.Text, nil
		}
		msgs = append(msgs, Message{Role: "assistant", ToolCalls: resp.ToolCalls})
		for _, call := range resp.ToolCalls {
			result := execTool(ctx, deps, mb, call)
			msgs = append(msgs, Message{Role: "tool", ToolName: call.Name, ToolID: call.ID, Content: result})
		}
	}
	return "", fmt.Errorf("agent exceeded %d iterations without final answer", maxIters)
}
